SaaS Auditoria
-- Tabla de Auditoría Específica para la Administración del SaaS CREATE TABLE saas_auditoria ( id SERIAL PRIMARY KEY, tabla_afectada VARCHAR(50) NOT NULL, operacion VARCHAR(10) NOT NULL, -- INSERT, UPDATE, DELETE registro_id INTEGER, usuario_admin VARCHAR(100) NOT NULL, datos_anteriores JSONB, datos_nuevos JSONB, fecha_evento TIMESTAMP DEFAULT CURRENT_TIMESTAMP, ip_origen VARCHAR(50) ); CREATE INDEX idx_saas_audit_tabla ON saas_auditoria(tabla_afectada); CREATE INDEX idx_saas_audit_fecha ON saas_auditoria(fecha_evento);
-- Trigger Genérico de Auditoría para el Módulo SaaS CREATE OR REPLACE FUNCTION registrar_auditoria_saas() RETURNS TRIGGER AS  BEGIN IF (TG_OP = 'DELETE') THEN INSERT INTO saas_auditoria (tabla_afectada, operacion, registro_id, usuario_admin, datos_anteriores) VALUES (TG_TABLE_NAME, 'DELETE', OLD.id, COALESCE(current_setting('app.current_admin_email', true), 'SUPER_ADMIN'), to_jsonb(OLD)); RETURN OLD; ELSIF (TG_OP = 'UPDATE') THEN INSERT INTO saas_auditoria (tabla_afectada, operacion, registro_id, usuario_admin, datos_anteriores, datos_nuevos) VALUES (TG_TABLE_NAME, 'UPDATE', NEW.id, COALESCE(current_setting('app.current_admin_email', true), 'SUPER_ADMIN'), to_jsonb(OLD), to_jsonb(NEW)); RETURN NEW; ELSIF (TG_OP = 'INSERT') THEN INSERT INTO saas_auditoria (tabla_afectada, operacion, registro_id, usuario_admin, datos_nuevos) VALUES (TG_TABLE_NAME, 'INSERT', NEW.id, COALESCE(current_setting('app.current_admin_email', true), 'SUPER_ADMIN'), to_jsonb(NEW)); RETURN NEW; END IF; RETURN NULL; END;  LANGUAGE plpgsql;
-- Aplicar Triggers de Auditoría en Tablas Contables/SaaS Críticas CREATE TRIGGER audit_saas_facturas_trigger AFTER INSERT OR UPDATE OR DELETE ON saas_facturas FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_saas();
CREATE TRIGGER audit_saas_pagos_trigger AFTER INSERT OR UPDATE OR DELETE ON saas_pagos FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_saas();
CREATE TRIGGER audit_saas_listas_precios_trigger AFTER INSERT OR UPDATE OR DELETE ON saas_listas_precios FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_saas();
