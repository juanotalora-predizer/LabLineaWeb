-- Tabla Diagnóstico / Resultados de Laboratorio (INMUTABLE) CREATE TABLE diagnostico ( consecutivo SERIAL PRIMARY KEY, tenant_id VARCHAR(50) NOT NULL, analisis_por_orden INTEGER NOT NULL, pregunta_por_analisis INTEGER NOT NULL, -- Referencia al detalle o parámetro del examen respuesta TEXT, chequeo VARCHAR(2) DEFAULT 'NO', usuario_digitacion VARCHAR(50) NOT NULL, fecha_digitacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP, hora_digitacion TIME DEFAULT CURRENT_TIME, CONSTRAINT fk_diag_apo FOREIGN KEY (analisis_por_orden) REFERENCES analisis_por_orden(consecutivo) ); CREATE INDEX idx_diag_tenant ON diagnostico(tenant_id); CREATE INDEX idx_diag_apo ON diagnostico(analisis_por_orden);
-- TRIGGER DE SEGURIDAD: Impide actualizaciones o borrados en Diagnóstico CREATE OR REPLACE FUNCTION prevenir_modificacion_diagnostico() RETURNS TRIGGER AS  BEGIN IF (TG_OP = 'UPDATE') THEN RAISE EXCEPTION 'Seguridad Médica: Los resultados de laboratorio (diagnostico) son inmutables y no pueden ser modificados. Debe emitir una adición o nota aclaratoria.'; ELSIF (TG_OP = 'DELETE') THEN RAISE EXCEPTION 'Seguridad Médica: Los resultados de laboratorio (diagnostico) no pueden ser eliminados.'; END IF; RETURN NEW; END;  LANGUAGE plpgsql;
CREATE TRIGGER trg_inmutabilidad_diagnostico BEFORE UPDATE OR DELETE ON diagnostico FOR EACH ROW EXECUTE FUNCTION prevenir_modificacion_diagnostico();
