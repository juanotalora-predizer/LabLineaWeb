NewTab Auditoria -- Tabla Central de Auditoría de Cambios Sensibles CREATE TABLE auditoria_sensible ( id SERIAL PRIMARY KEY, tenant_id VARCHAR(50) NOT NULL, tabla_afectada VARCHAR(50) NOT NULL, operacion VARCHAR(10) NOT NULL, -- INSERT, UPDATE, DELETE registro_id INTEGER, usuario VARCHAR(50) NOT NULL, datos_anteriores JSONB, datos_nuevos JSONB, fecha_evento TIMESTAMP DEFAULT CURRENT_TIMESTAMP, ip_origen VARCHAR(50) ); CREATE INDEX idx_audit_tenant_tabla ON auditoria_sensible(tenant_id, tabla_afectada); CREATE INDEX idx_audit_fecha ON auditoria_sensible(fecha_evento);
-- Función genérica para el Audit Trigger CREATE OR REPLACE FUNCTION registrar_auditoria_cambios() RETURNS TRIGGER AS  BEGIN IF (TG_OP = 'DELETE') THEN INSERT INTO auditoria_sensible (tenant_id, tabla_afectada, operacion, registro_id, usuario, datos_anteriores) VALUES (OLD.tenant_id, TG_TABLE_NAME, 'DELETE', OLD.consecutivo, COALESCE(current_setting('app.current_user_name', true), 'SISTEMA'), to_jsonb(OLD), NULL); RETURN OLD; ELSIF (TG_OP = 'UPDATE') THEN INSERT INTO auditoria_sensible (tenant_id, tabla_afectada, operacion, registro_id, usuario, datos_anteriores, datos_nuevos) VALUES (NEW.tenant_id, TG_TABLE_NAME, 'UPDATE', NEW.consecutivo, COALESCE(current_setting('app.current_user_name', true), 'SISTEMA'), to_jsonb(OLD), to_jsonb(NEW)); RETURN NEW; ELSIF (TG_OP = 'INSERT') THEN INSERT INTO auditoria_sensible (tenant_id, tabla_afectada, operacion, registro_id, usuario, datos_nuevos) VALUES (NEW.tenant_id, TG_TABLE_NAME, 'INSERT', NEW.consecutivo, COALESCE(current_setting('app.current_user_name', true), 'SISTEMA'), to_jsonb(NEW)); RETURN NEW; END IF; RETURN NULL; END;  LANGUAGE plpgsql;
-- Aplicar Triggers de Auditoría a Tablas Críticas CREATE TRIGGER audit_paciente_trigger AFTER INSERT OR UPDATE OR DELETE ON paciente FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_cambios();
CREATE TRIGGER audit_orden_trigger AFTER INSERT OR UPDATE OR DELETE ON orden FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_cambios();
CREATE TRIGGER audit_diagnostico_trigger AFTER INSERT ON diagnostico -- Solo insert ya que UPDATE/DELETE están bloqueados FOR EACH ROW EXECUTE FUNCTION registrar_auditoria_cambios();
